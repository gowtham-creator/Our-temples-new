    ```tsx
    import './globals.css';

    export const metadata: Metadata = {
      title: 'Sonic',
      description: 'Premium speaker experience',
    };

    export default function RootLayout({
      children,
    }: Readonly<{
      children: React.ReactNode;
    }>) {
      return (
        <html lang="en" className={inter.variable}>
          <body>{children}</body>
        </html>
      );
    }
    ```
